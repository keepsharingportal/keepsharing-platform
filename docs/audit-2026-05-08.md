```ts
  category: string
  categoryHref?: string
  publishedDate: string
  readTimeMinutes: number

}
```

### ArticleBody.tsx
```ts
interface Props {
  body: string
  pullQuotes?: string[]
  inlineAd?: ReactNode
  inlineCta?: ReactNode
}
```
Renders DOMPurify-sanitized HTML. Injects pull quotes + ad/CTA at computed offsets.

```ts
interface Props {
  stickyAd?: AdPlacement | null
  sponsoredAd?: AdPlacement | null

}
```

```ts

  headline: string; description: string
  ctaLabel: string; ctaUrl: string
}
```

### ArticleAuthorBlock.tsx
```ts
interface Props {
  authorName: string
  authorRole?: string
  authorAvatarUrl?: string | null
  shareUrl: string
}
```
Detects "River Region Parents" / "RRP Editorial" / "Editorial" for special styling.
